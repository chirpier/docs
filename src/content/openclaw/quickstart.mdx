# OpenClaw Quickstart

This is the fastest path to get ChirrOp working with OpenClaw.

## Recommended path

1. create or copy a ChirrOp API key from the web app
2. export that key into the OpenClaw runtime as `CHIRROP_API_KEY`
3. install or load `chirrop-skills`
4. start with the `chirrop` skill
5. emit one canonical OpenClaw event
6. confirm the event definition appears in ChirrOp
7. create one chart and one policy

For v1, ChirrOp standardizes backend rollup periods on:

- `minute`
- `hour`
- `day`

Use those values in event-log and alerting APIs. UI shortcuts like `30d` belong to chart/embed filtering, not the backend period contract.

If you want the opinionated OpenClaw workflow, use `chirrop-skills`. If you want the thinnest possible setup, emit the same events directly with the SDK.

```ts
import { createClient } from "@chirropjs/chirrop-js";

const client = createClient({ key: process.env.CHIRROP_API_KEY! });

await client.log({
  agent: "openclaw.main",
  event: "tool.errors.count",
  value: 1,
  meta: { tool_name: "browser.open", workflow: "triage" },
});

await client.flush();

const events = await client.listEvents();
const toolErrors = events.find(
  (eventDef) => eventDef.agent === "openclaw.main" && eventDef.event === "tool.errors.count",
);

if (toolErrors) {
  await client.getEventAnalytics(toolErrors.event_id, {
    view: "window",
    period: "1h",
    previous: "previous_window",
  });

  await client.createPolicy({
    event_id: toolErrors.event_id,
    title: "OpenClaw tool errors spike",
    channel: "ops",
    period: "hour",
    aggregate: "sum",
    condition: "gt",
    threshold: 5,
    severity: "warning",
    enabled: true,
  });
}

await client.shutdown();
```

```python
from chirrop import Log, new_client

client = new_client(api_key="chp_your_api_key")
client.log(Log(agent="openclaw.main", event="tool.errors.count", value=1, meta={"tool_name": "browser.open", "workflow": "triage"}))
client.flush()

events = client.list_events()
tool_errors = next(
    (event_def for event_def in events if event_def.get("agent") == "openclaw.main" and event_def.get("event") == "tool.errors.count"),
    None,
)

if tool_errors:
    client.get_event_analytics(tool_errors["event_id"], view="window", period="1h", previous="previous_window")

client.close()
```

```go
ctx := context.Background()
client, err := chirrop.NewClient(chirrop.Options{Key: "chp_your_api_key"})
if err != nil {
    return err
}
defer client.Close(ctx)

_ = client.Log(ctx, chirrop.Log{Agent: "openclaw.main", Event: "tool.errors.count", Value: 1, Meta: map[string]any{"tool_name": "browser.open", "workflow": "triage"}})
_ = client.Flush(ctx)

events, _ := client.ListEvents(ctx)
for _, eventDef := range events {
    if eventDef.Agent == "openclaw.main" && eventDef.Event == "tool.errors.count" {
        _, _ = client.GetEventAnalytics(ctx, eventDef.EventID, chirrop.AnalyticsWindowQuery{View: "window", Period: "1h", Previous: "previous_window"})
    }
}
```
